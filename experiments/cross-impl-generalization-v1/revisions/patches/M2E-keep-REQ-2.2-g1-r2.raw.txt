export async function editorState(editor: Locator): Promise<EditorObs> {
  // Use both visibility and DOM presence to detect editor state
  const visible = await editor.isVisible().catch(() => false);
  const exists = await editor.count().catch(() => 0) > 0;
  const open = visible && exists;
  return {
    open,
    evidence: open
      ? 'editor locator still visible and present in DOM'
      : `editor ${!exists ? 'not present in DOM' : 'not visible'}`
  };
}
