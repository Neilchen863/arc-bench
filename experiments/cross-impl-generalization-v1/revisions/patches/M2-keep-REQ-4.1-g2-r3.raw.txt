// In-browser: locate note records by their exact visible title and return the record's visual lines.
// A record is the nearest ancestor of the title element that is

export async function suggestionPanel(page: Page): Promise<{ shown: boolean; evidence: string; panel: Locator | null }> {
  // Retry up to 5 times with 250ms intervals to handle slight delays in panel appearance

    const buttonBased = page.locator('[role="listbox"],[role="group"],[role="menu"]').filter({
      has: page.getByRole('button', { name: /Reminders/i })
    }).first();
    if (await buttonBased.isVisible()) {
      return { shown: true, evidence: 'suggestion panel found by Reminders button', panel: buttonBased };
    }

    // Wait before retrying
    if (attempt < 4) await new Promise(r => setTimeout(r, 250));
  }

  return { shown: false, evidence: 'no suggestion container found after retries', panel: null };
